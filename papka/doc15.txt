2. Главные проблемы
Критические
1. Монолитный JS файл ~2000 строк невозможно поддерживать, нет модульности, весь код в IIFE
2. CSP дублируется и конфликтует задан одновременно в _headers, .htaccess, nginx.conf и <meta> теге с разными значениями
3. Контактная форма не подключена только event.preventDefault() + toast, нет реальной отправки, нет CSRF токена, нет rate limiting
4. Нет реальной защиты от спама форма может быть использована для спаматак
5. Canvas работает постоянно даже когда пользователь не взаимодействует, потребляет CPU/GPU на мобильных
Важные
6. Нет lazy loading для фоновых изображений карточек все загружаются сразу
7. Нет preload для критических ресурсов шрифты, ключевые изображения
8. Нет fallback для отключенного JS сайт без JS практически пустой
9. Нет sitemap.xml, robots.txt SEO страдает
10. Нет Open Graph / Twitter Card метатегов ссылки в соцсетях будут без превью
Средние
11. Дублирование контента секции "Studio Philosophy", "What We Build", "About" повторяют друг друга
12. Нет 404 страницы
13. Нет страницы игры модальное окно вместо отдельной страницы (плохо для SEO)
14. Шрифт Inter не подгружается только через system fonts fallback
15. Нет аналитики нет ни Google Analytics, ни Plausible, ни собственной

 let timer;
 return (...args) => {
 clearTimeout(timer);
 timer = setTimeout(() => fn(...args), ms);
 };
}

const cssCache = new Map();
function css(name) {

Проблемы и рекомендации:
1. CSP дублируется и конфликтует:
ПРОБЛЕМА: CSP задан в 4 местах с разными значениями:
_headers (Netlify/Vercel)
.htaccess (Apache)
nginx.conf (Nginx)
<meta> тег (fallback)
РЕКОМЕНДАЦИЯ: Выбрать один способ для вашего хостинга.
Для Nginx только nginx.conf, убрать metaтег.
Для Netlify только _headers, убрать metaтег.
Metaтег CSP это fallback, но он может конфликтовать с header CSP.
Браузер применяет наиболее строгий из них.
2. Контактная форма без защиты:
ПРОБЛЕМА: Форма только на клиенте, нет серверной валидации
ПРОБЛЕМА: Нет CSRF токена
ПРОБЛЕМА: Нет rate limiting
ПРОБЛЕМА: Нет honeypot для ботов
РЕКОМЕНДАЦИЯ:
<!Добавить honeypot >
<div style="position:absolute;left:9999px" ariahidden="true">
 <input type="text" name="website" tabindex="1" autocomplete="off">
</div>
<!Добавить CSRF токен (генерируется сервером) >
<input type="hidden" name="csrf_token" value="{{csrf_token}}">
<!На сервере: >
// 1. Проверить CSRF токен
// 2. Валидировать все поля
// 3. Rate limit: max 5 запросов с IP в час
// 4. Санитизировать вывод
// 5. Отправить через транзакционный email (SendGrid, Postmark)

class RingBuffer {
 constructor(size) {
 this.size = size;
 this.buffer = new Array(size);
 this.index = 0;
 this.length = 0;
 }
 
 push(item) {
 this.buffer[this.index] = item;
 this.index = (this.index + 1) % this.size;
 this.length = Math.min(this.length + 1, this.size);
 }
 
 [Symbol.iterator]() {
 for (let i = 0; i < this.length; i++) {
   const idx = (this.index this.length + i + this.size) % this.size;

 }
 }
}

   ctx.globalAlpha = alpha;
   ctx.moveTo(star.x + star.r, star.y);
   ctx.arc(star.x, star.y, star.r, 0, Math.PI 2);
 }
 
 ctx.fill();
 ctx.globalAlpha = 1;
 }

 const base = dark ? 0.26 : 0.12;
 const twinkle = Math.sin(time 0.002 star.depth + star.twinkle) 0.2;
 const depthBonus = star.depth (dark ? 0.28 : 0.18);
 return Math.max(dark ? 0.12 : 0.06, Math.min(dark ? 0.92 : 0.48, base + twinkle + depthBonus));
 }
}

class ParticleSystem {
 constructor(maxCount) {
 this.count = 0;
 this.maxCount = maxCount;
 
 // SoA (Structure of Arrays) для кэшэффективности
 this.x = new Float32Array(maxCount);
 this.y = new Float32Array(maxCount);
 this.vx = new Float32Array(maxCount);
 this.vy = new Float32Array(maxCount);
 this.r = new Float32Array(maxCount);
 this.hue = new Float32Array(maxCount);
 this.active = new Uint8Array(maxCount);
 }
 
 add(x, y, vx, vy, r, hue) {
 if (this.count >= this.maxCount) return;
 
 const i = this.count++;
 this.x[i] = x;
 this.y[i] = y;
 this.vx[i] = vx;
 this.vy[i] = vy;
 this.r[i] = r;
 this.hue[i] = hue;
 this.active[i] = 1;
 }
 
 update(dt, width, height, time) {
 const dtFactor = dt / 16.67;
 
 for (let i = 0; i < this.count; i++) {
   if (!this.active[i]) continue;
   
   this.x[i] += this.vx[i] dtFactor;
   this.y[i] += this.vy[i] dtFactor;
   this.x[i] += Math.sin(time 0.001 + this.y[i] 0.01) 0.04 dtFactor;
   
   // Wrap around
   if (this.y[i] < 20) {
     this.y[i] = height + 20;
     this.x[i] = Math.random() width;
   }
   if (this.x[i] < 20) this.x[i] = width + 20;
   if (this.x[i] > width + 20) this.x[i] = 20;
 }
 }
 
 draw(ctx, colors) {
 ctx.globalAlpha = 0.18;
 
 for (let i = 0; i < this.count; i++) {
   if (!this.active[i]) continue;
   
   ctx.globalAlpha = 0.18 + this.hue[i] 0.34;
   ctx.fillStyle = colors[Math.floor(this.hue[i] colors.length) % colors.length];
   ctx.beginPath();
   ctx.arc(this.x[i], this.y[i], this.r[i], 0, Math.PI 2);
   ctx.fill();
 }
 
 ctx.globalAlpha = 1;
 }
}

class PerformanceMonitor {

 }

 // Мониторинг Long Tasks
 if ('PerformanceObserver' in window) {
   try {
     this.longTaskObserver = new PerformanceObserver(list => {
       for (const entry of list.getEntries()) {
         if (entry.duration > 50) {
           this.metrics.longTasks++;
           console.warn(Long task detected: ${entry.duration.toFixed(1)}ms);
         }
       }
     });

   } catch (e) {
     // longtask не поддерживается
   }
 }
 }
 
 recordFrame(deltaTime) {
 this.metrics.frameTime.push(deltaTime);
 this.metrics.fps.push(1000 / deltaTime);
 
 // Хранить только последние 60 кадров
 if (this.metrics.frameTime.length > 60) {
   this.metrics.frameTime.shift();
   this.metrics.fps.shift();
 }
 }
 
 getAverageFPS() {
 if (this.metrics.fps.length === 0) return 0;
 return this.metrics.fps.reduce((a, b) => a + b, 0) / this.metrics.fps.length;
 }
 
 getAverageFrameTime() {
 if (this.metrics.frameTime.length === 0) return 0;
 return this.metrics.frameTime.reduce((a, b) => a + b, 0) / this.metrics.frameTime.length;
 }
 
 getMemoryInfo() {
 if (performance.memory) {

 }

 }

class RateLimiter {
 constructor(maxRequests = 5, windowMs = 3600000) { // 5 запросов в час
 this.maxRequests = maxRequests;
 this.windowMs = windowMs;
 this.storageKey = 'rate_limit_requests';
 }
 
 getRequests() {
 try {
   const data = localStorage.getItem(this.storageKey);
   if (!data) return [];
   
   const requests = JSON.parse(data);
   const now = Date.now();
   
   // Очистить старые запросы
   return requests.filter(time => now time < this.windowMs);
 } catch (e) {
   return [];
 }
 }
 
 canMakeRequest() {
 const requests = this.getRequests();
 return requests.length < this.maxRequests;
 }
 
 recordRequest() {
 const requests = this.getRequests();
 requests.push(Date.now());
 
 try {
   localStorage.setItem(this.storageKey, JSON.stringify(requests));
 } catch (e) {
   // Storage full или недоступен
 }
 }
 
 getRemainingRequests() {
 return Math.max(0, this.maxRequests this.getRequests().length);
 }
 
 getTimeUntilReset() {
 const requests = this.getRequests();
 if (requests.length === 0) return 0;
 
 const oldestRequest = Math.min(...requests);
 const resetTime = oldestRequest + this.windowMs;
 return Math.max(0, resetTime Date.now());
 }
}

 return;
 }
 
 // ... отправка формы
 formRateLimiter.recordRequest();
}

// Улучшенная ловушка фокуса для модальных окон
class FocusTrap {
 constructor(element) {
 this.element = element;
 this.focusableSelectors = [
   'a[href]:not([disabled]):not([tabindex="1"])',
   'button:not([disabled]):not([tabindex="1"])',
   'input:not([disabled]):not([tabindex="1"])',
   'select:not([disabled]):not([tabindex="1"])',
   'textarea:not([disabled]):not([tabindex="1"])',
   '[tabindex]:not([tabindex="1"])'
 ].join(', ');
 
 this.handleKeyDown = this.handleKeyDown.bind(this);
 }
 
 getFocusableElements() {
 return Array.from(
   this.element.querySelectorAll(this.focusableSelectors)
 ).filter(el => {
   // Проверить видимость
   const style = window.getComputedStyle(el);
   return style.display !== 'none' && 
          style.visibility !== 'hidden' &&
          el.offsetParent !== null;
 });
 }
 
 handleKeyDown(event) {
 if (event.key !== 'Tab') return;
 
 const focusable = this.getFocusableElements();
 if (focusable.length === 0) return;
 
 const first = focusable[0];
 const last = focusable[focusable.length 1];
 
 if (event.shiftKey) {
   if (document.activeElement === first) {
     event.preventDefault();
     last.focus();
   }
 } else {
   if (document.activeElement === last) {
     event.preventDefault();
     first.focus();
   }
 }
 }
 
 activate() {
 document.addEventListener('keydown', this.handleKeyDown);
 // Фокус на первый элемент
 const focusable = this.getFocusableElements();
 if (focusable.length > 0) {
   focusable[0].focus();
 }
 }
 
 deactivate() {
 document.removeEventListener('keydown', this.handleKeyDown);
 }
}

Защита API endpoints
javascript
// Rate limiting для API (Cloudflare Workers / Node.js)
const rateLimit = new Map();
function checkRateLimit(ip, maxRequests = 10, windowMs = 60000) {
 const now = Date.now();
 const key = ip;
 
 if (!rateLimit.has(key)) {
 rateLimit.set(key, { count: 1, resetTime: now + windowMs });
 return { allowed: true, remaining: maxRequests 1 };
 }
 
 const record = rateLimit.get(key);
 
 if (now > record.resetTime) {
 record.count = 1;
 record.resetTime = now + windowMs;
 return { allowed: true, remaining: maxRequests 1 };
 }
 
 record.count++;
 
 if (record.count > maxRequests) {
 return { 
   allowed: false, 
   remaining: 0,
   retryAfter: Math.ceil((record.resetTime now) / 1000)
 };
 }
 
 return { allowed: true, remaining: maxRequests record.count };
}
// Middleware для Express
function rateLimitMiddleware(req, res, next) {
 const ip = req.ip  req.connection.remoteAddress;
 const result = checkRateLimit(ip, 10, 60000); // 10 запросов в минуту
 
 res.set({
 'XRateLimitLimit': 10,
 'XRateLimitRemaining': result.remaining,
 'XRateLimitReset': new Date(Date.now() + 60000).toISOString()
 });
 
 if (!result.allowed) {
 return res.status(429).json({
   error: 'Too many requests',
   retryAfter: result.retryAfter
 });
 }
 
 next();
}

2. Новые возможные причины проблем
Возможная причина  Почему это проблема  Как проверить  Вероятность  Опасность  Приоритет 
getComputedStyle каждый кадр Forced style recalculation каждый кадр = 25ms overhead. На 60fps = 1230% времени кадра только на это  Chrome DevTools → Performance → искать "Recalculate Style"  95%  Высокая  🔴 P0 
Canvas без contain Каждый repaint canvas может вызывать repaint всей страницы  DevTools → Rendering → Paint flashing  80%  Высокая  🔴 P0 
Нет willchange Браузер не знает какие элементы будут анимироваться, не создаёт отдельные слои  DevTools → Layers → проверить слои  70%  Средняя  🟠 P1 
Touch events конфликт Click listener на window может задерживать скролл на мобильных  Тестирование на реальном устройстве  60%  Средняя  🟠 P1 
Изображения без preload LCP увеличивается на 200500ms  Network tab → порядок загрузки  90%  Высокая  🔴 P0 
Нет critical CSS Весь CSS блокирует рендеринг  Coverage tab → сколько CSS реально используется  85%  Средняя  🟠 P1 
Дублирование данных HTML/JS Рассинхронизация при обновлении  Ручная проверка  50%  Средняя  🟡 P2 
Нет обработки ошибок изображений Сломанные изображения без fallback  Отключить сеть → проверить  40%  Низкая  🟡 P2 
Canvas без адаптации под устройство Лаги на слабых устройствах  Тестирование на бюджетном Android  75%  Высокая  🔴 P0 
Нет reduced motion для canvas Анимации нельзя отключить  System settings → Reduce motion  100%  Средняя  🟠 P1 
Модалки без URL Нельзя шарить, нельзя индексировать  Проверить history API  100%  Средняя  🟠 P1 
Нет CSRF токена Форма уязвима для CSRF атак  Попробовать отправить форму с другого домена  100%  Высокая  🔴 P0 
Нет rate limiting Форма может быть использована для спама  Отправить 100 запросов подряд  100%  Высокая  🔴 P0 
Нет CSP report endpoint Нельзя мониторить нарушения CSP  Проверить network при нарушении  100%  Низкая  🟡 P2 
Нет обработки canvas null Код упадёт если getContext вернёт null  Тестировать в старом браузере  20%  Средняя  🟡 P2 
3. Новые способы фикса
Способ фикса  Что изменить  Почему поможет  Риски  Когда использовать  Productionready? 
Кеширование CSS переменных Заменить css() функцию на объект с кешем  Убирает forced recalculation каждый кадр, экономит 25ms на кадр  Нужно инвалидировать кеш при смене темы  Всегда для canvasанимаций  ✅ Да 
CSS contain для секций Добавить contain: layout style paint на .section  Изолирует repaint, предотвращает каскадные reflow  Может сломать position: fixed внутри  Для секций с анимациями  ✅ Да 
willchange + отключение Добавить willchange: transform перед анимацией, убрать после  Создаёт отдельный слой GPU, ускоряет compositing  Чрезмерное использование = утечка памяти  Только для элементов с частой анимацией  ✅ Да 
Critical CSS inline Инлайнить стили для первого экрана, остальной CSS async  Уменьшает FCP на 200500ms  Усложняет поддержку  Для всех production сайтов  ✅ Да 
Preload hero изображений Добавить <link rel="preload"> для LCP изображений  Ускоряет LCP на 200400ms  Не перегружать максимум 23 preload  Для критических изображений  ✅ Да 
IntersectionObserver rootMargin Добавить rootMargin: "200px 0px" для раннего старта анимаций  Анимации начинаются раньше, пользователь видит результат  Может загрузить больше контента  Для scrolltriggered анимаций  ✅ Да 
Passive event listeners Добавить { passive: true } для touch/wheel  Убирает задержку скролла на мобильных  Нельзя использовать preventDefault  Для всех touch listeners  ✅ Да 
Debounced resize Debounce resize handler до 150ms  Уменьшает количество layout calculations  Небольшая задержка при resize  Всегда  ✅ Да 
Image error fallback Добавить onerror для всех изображений  Предотвращает сломанный UI  Нужен fallback контент  Всегда  ✅ Да 
Canvas feature detection Проверить getContext перед использованием  Предотвращает crash в старых браузерах  Нужен fallback  Всегда  ✅ Да 
URL для модалок Использовать history.pushState при открытии  Позволяет шарить, навигировать назад  Нужна обработка popstate  Для всех модальных окон  ✅ Да 
CSRF токен Генерировать на сервере, проверять при отправке  Защита от CSRF атак  Нужен backend  Для всех форм  ✅ Да 
Rate limiting Ограничить 5 запросов в час на IP  Защита от спама  Может заблокировать легитимных пользователей  Для всех API endpoints  ✅ Да 
Adaptive quality Проверить deviceMemory, hardwareConcurrency при старте  Превентивное снижение кастомизации  Может занизить качество на хороших устройствах  Для canvasанимаций  ✅ Да 
CSS prefersreduceddata Добавить медиазапрос для reduceddata  Уважает настройки пользователя  Нужен fallback  Всегда  ✅ Да 
4. Сравнение подходов
4.1. Оптимизация Canvas
Подход  Надёжность  Сложность  Performance  UX  SEO  Security  Риск багов  Production 
Кеширование CSS переменных Высокая  Низкая  +++  =  =  =  Низкий  ✅ 
OffscreenCanvas (Worker) Высокая  Высокая  ++++  =  =  =  Средний  ⚠️ 
Автоматическое снижение FPS Средняя  Средняя  ++  +  =  =  Низкий  ✅ 
Полное отключение на слабых Высокая  Низкая  +++   =  =  Низкий  ✅ 
WebGL вместо 2D Canvas Средняя  Очень высокая  ++++  =  =  =  Высокий  ⚠️ 
Вывод:Кеширование CSS + адаптивное качество лучший баланс. OffscreenCanvas для будущего. WebGL избыточно.

function validateField(field) {
 const value = field.value.trim();
 const errorEl = field.parentElement.querySelector('.fielderror');
 
 let error = '';
 
 if (field.type === 'email' && value) {
 const emailRe = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
 if (!emailRe.test(value)) {
   error = 'Please enter a valid email';
 }
 }
 
 if (field.hasAttribute('required') && !value) {
 error = 'This field is required';
 }
 
 if (errorEl) {
 errorEl.textContent = error;
 }
 
 field.classList.toggle('isinvalid', !!error);
 field.classList.toggle('isvalid', !error && value);
 
 return !error;
}

function detectCapabilities() {
 const memory = navigator.deviceMemory  4;
 const cores = navigator.hardwareConcurrency  4;
 const connection = navigator.connection  {};
 
 // Проверить все факторы
 const isLowEnd = memory < 4  cores < 4;
 const isSlowNetwork = /^(slow2g2g3g)$/.test(connection.effectiveType  '');
 const saveData = connection.saveData  false;
 const reducedMotion = window.matchMedia('(prefersreducedmotion: reduce)').matches;
 const reducedData = window.matchMedia('(prefersreduceddata: reduce)').matches;
 
 if (reducedMotion  saveData  reducedData) {
 return { quality: 'minimal', particles: 0, fps: 0 };
 }
 
 if (isLowEnd  isSlowNetwork) {
 return { quality: 'low', particles: 15, fps: 15 };
 }
 
 if (memory < 8  cores < 6) {
 return { quality: 'medium', particles: 35, fps: 30 };
 }
 
 return { quality: 'high', particles: 70, fps: 60 };
}
